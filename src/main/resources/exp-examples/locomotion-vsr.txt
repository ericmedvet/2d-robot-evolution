$basePath = ''../../Documenti/experiments/{^^.name}/{^^.startTime}/''

ea.experiment(
  runs = (randomGenerator = (seed = [1:1:3]) * [m.defaultRG()]) *
    (solver = (mapper = [
      er.m.ndsToFixedBodyCentralizedVsr(
        name = "centralized-mlp";
        body = s.a.vsr.gridBody(
          sensorizingFunction = s.a.vsr.sf.directional(
            headSensors = [s.s.sin(f = 0); s.s.d(a = -15; r = 5)];
            nSensors = [s.s.ar(); s.s.rv(a = 0); s.s.rv(a = 90)];
            sSensors = [s.s.d(a = -90)]
          );
          shape = s.a.vsr.s.biped(w = 4; h = 3)
        );
        of = ea.m.noisedNds(
          of = ea.m.steppedNds(of = ea.m.dsToNpnds(npnds = ds.num.mlp()); stepT = 0.2);
          randomGenerator = m.sharedRG()
        )
      );
      er.m.ndsToFixedBodyHomoDistributedVsr(
        name = "distributed-mlp";
        body = s.a.vsr.gridBody(
          sensorizingFunction = s.a.vsr.sf.uniform(
            sensors = [s.s.ar(); s.s.rv(a = 0); s.s.rv(a = 90)]
          );
          shape = s.a.vsr.s.biped(w = 4; h = 3)
        );
        of = ea.m.noisedNds(
          of = ea.m.steppedNds(of = ea.m.dsToNpnds(npnds = ds.num.mlp()); stepT = 0.2);
          randomGenerator = m.sharedRG()
        )
      )
    ]) * [
      ea.s.ga(
        representation = ea.r.doubleString();
        stop = ea.sc.nOfQualityEvaluations(n = 2000);
        nPop = 50
      )
    ]) * [
    ea.run(
      problem = ea.p.simToSmo(
        simulation = s.task.locomotion();
        toMaxObjectives = [s.f.outcome.faXVelocity()];
        dT = 0.016666;
        tRange = m.range(min = 0; max = 10)
      )
    )
  ];
  listeners = [
    ea.l.console(
      eFunctions = [
        ea.f.size(of = ea.f.genotype(of = ea.f.best()); format = "%3d");
        f.mapValue(key = "first.agent.velocity.x"; of = ea.f.quality(of = ea.f.best()); format = "%5.3f");
        ea.f.hist(of = f.each(of = ea.f.all(); mapF = f.mapValue(key = "first.agent.velocity.x"; of = ea.f.quality())))
      ];
      kFunctions = [
        f.mappableKey(key = "solver.mapper.name")
      ]
    );
    ea.l.saveForRun(
      path = $basePath + "{solver.mapper.name}/behavior-{randomGenerator.seed:%02d}.mp4";
      of = ea.acc.lastBest();
      processor = s.f.taskVideo(
        of = ea.f.solution();
        endTime = 10;
        task = s.task.locomotion()
      )
    );
    ea.l.savePlotForExp(
      path = $basePath + "best-fitness.svg";
      plot = ea.plot.multi.quality(
        q = f.mapValue(key = "first.agent.velocity.x");
        line = f.interpolated(name = solver; s = "{solver.mapper.name}")
      );
      type = svg
    )
  ]
)
