$SIMULATION = s.task.sumoCup(duration = 15)

$VS_SELF = ds.f.selfBiSimulator(
  of = ea.f.solution();
  simulation = $SIMULATION
)

$VS_OPPONENT = ds.f.opponentBiSimulator(
  of = ea.f.solution();
  opponent = m.supplier(of = s.a.dummyBox());
  simulation = $SIMULATION
)

$MAPPER = er.m.bodyBrainHomoDistributedVSR(
  w = 4;
  h = 4;
  sensors = [s.s.a(); s.s.ar(); s.s.rv(a = 0); s.s.rv(a = 90); s.s.d(r = 5)];
  of = ea.m.pair(
    of = ea.m.splitter();
    first = ea.m.dsToFixedGrid(negItem = s.a.vsr.voxel(type = none); posItem = s.a.vsr.voxel(type = soft));
    second = ea.m.steppedNds(of = ea.m.dsToNpnds(npnds = ds.num.mlp(innerLayerRatio = 2; nOfInnerLayers = 2)); stepT = 0.2)
  )
)

$BODY_DESCRIPTORS = [
  ea.s.me.d.descriptor(f = f.gridCount(of = s.f.vsrBody(of = ea.f.supplied(of = ea.f.solution()))); min = 1; max = 16; nOfBins = 10);
  ea.s.me.d.descriptor(f = f.gridElongation(of = s.f.vsrBody(of = ea.f.supplied(of = ea.f.solution()))); min = 0; max = 1; nOfBins = 10)
]

$THRESHOLD = 0.5
$NEG_THRESHOLD = -0.5

$BRAIN_EXTRACTOR = ds.f.inner(of = f.first(of = s.f.numBrains(of = ea.f.supplied(of = ea.f.solution()))))

$BRAIN_DESCRIPTORS = [
  ea.s.me.d.descriptor(
    f = f.mathOp(
      op = divide;
      args = [
        f.size(of = f.filter(of = ea.f.getLayerWeights(of = $BRAIN_EXTRACTOR; indexOfLayer = 0); condition = predicate.all(conditions = [predicate.lt(t = $THRESHOLD); predicate.gt(t = $NEG_THRESHOLD)])));
        f.size(of = ea.f.getLayerWeights(of = $BRAIN_EXTRACTOR; indexOfLayer = 0))
      ]
    );
    min = 0.0;
    max = 1.0;
    nOfBins = 10
  );
  ea.s.me.d.descriptor(
    f = f.mathOp(
      op = divide;
      args = [
        f.size(of = f.filter(of = ea.f.getLayerWeights(of = $BRAIN_EXTRACTOR; indexOfLayer = 1); condition = predicate.all(conditions = [predicate.lt(t = $THRESHOLD); predicate.gt(t = $NEG_THRESHOLD)])));
        f.size(of = ea.f.getLayerWeights(of = $BRAIN_EXTRACTOR; indexOfLayer = 1))
      ]
    );
    min = 0.0;
    max = 1.0;
    nOfBins = 10
  )
]
