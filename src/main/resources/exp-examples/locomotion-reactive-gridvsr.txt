$DURATION = 15

  w = 6;
  h = 4;
  availableVoxels = [
      m.supplier(of = s.a.vsr.rv.ph());
      m.supplier(of = s.a.vsr.rv.ps());
      m.supplier(of = s.a.vsr.rv.avsin());
      m.supplier(of = s.a.vsr.rv.ahsin())
  ]
)
$GRAMMAR_MAPPER = er.m.sGridToReactiveGridVsr(
  of = ea.m.bsToGrammarGrid(
    l = 1024;
    grammar = ea.grammar.gridFile(path = "src/main/resources/grammars/biped-vh123.bnf")
  );
  availableVoxels = ea.misc.sMapFromLists(
    keys = [h; s; avsin; ahsin];
    values = [
      m.supplier(of = s.a.vsr.rv.ph());
      m.supplier(of = s.a.vsr.rv.ps());
      m.supplier(of = s.a.vsr.rv.avsin());
      m.supplier(of = s.a.vsr.rv.ahsin())
    ]
  )
)
