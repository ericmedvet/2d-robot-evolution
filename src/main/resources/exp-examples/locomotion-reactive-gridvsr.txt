ea.experiment(
  runs = (randomGenerator = (seed = [1:1:1]) * [m.defaultRG()]) *
    (problem = [
      ea.p.simToSbbmo(
        simulation = s.task.locomotion(duration = 15);
        toMaxObjectives = [s.f.outcome.faXVelocity()]
      )
    ]) *
    (solver = (nEval = [500]) * (nPop = [10]) * [
      ea.s.ga(
        representation = ea.r.intString();
        mapper = er.m.isToReactiveGridVsr(
          w = 5;
          h = 5;
          availableVoxels = [
            m.supplier(of = s.a.vsr.rv.aa());
            m.supplier(of = s.a.vsr.rv.asin());
            m.supplier(of = s.a.vsr.rv.asld());
            m.supplier(of = s.a.vsr.rv.at());
            m.supplier(of = s.a.vsr.rv.ph());
            m.supplier(of = s.a.vsr.rv.ps())
          ]
        )
      );
      ea.s.ga(
        representation = ea.r.bitString();
        mapper = er.m.cGridToReactiveGridVsr(
          of = ea.m.bsToGrammarGrid(grammar = ea.grammar.gridBundled(name = "dog-shape"));
          availableVoxels = ea.misc.map(entries = [
            ea.misc.entry(key = ea.misc.ch(s = "s"); value = m.supplier(of = s.a.vsr.rv.ps()));
            ea.misc.entry(key = ea.misc.ch(s = "a"); value = m.supplier(of = s.a.vsr.rv.asin()));
            ea.misc.entry(key = ea.misc.ch(s = "h"); value = m.supplier(of = s.a.vsr.rv.ph()))
          ])
        )
      )
    ]) *
    [ea.run()];
  listeners = [
    ea.l.console(
      functions = [
        ea.f.size(of = ea.f.genotype(of = ea.f.best()); format = "%3d");
        ea.f.hist(of = f.each(of = ea.f.all(); mapF = f.mapValue(key = "first.agent.velocity.x"; of = ea.f.behaviorQuality(of = ea.f.quality()))));
        f.mapValue(key = "first.agent.velocity.x"; of = ea.f.behaviorQuality(of = ea.f.quality(of = ea.f.best())); format = "%6.4f");
        f.gridFitW(of = s.f.vsrBody(of = ea.f.supplied(of = ea.f.solution(of = ea.f.best()))));
        f.gridFitH(of = s.f.vsrBody(of = ea.f.supplied(of = ea.f.solution(of = ea.f.best()))));
        f.gridCount(of = s.f.vsrBody(of = ea.f.supplied(of = ea.f.solution(of = ea.f.best()))));
        f.gridCompactness(of = s.f.vsrBody(of = ea.f.supplied(of = ea.f.solution(of = ea.f.best()))); format = "%6.4f");
        f.gridElongation(of = s.f.vsrBody(of = ea.f.supplied(of = ea.f.solution(of = ea.f.best()))); format = "%6.4f")
      ];
      logExceptions = true
    );
    ea.l.savePlotForExp(
      path = "../../Documenti/experiments/{name}/{startTime}/best-fitness";
      plot = ea.plot.multi.quality(
        x = ea.f.nOfEvals();
        q = f.mapValue(key = "first.agent.velocity.x"; of = ea.f.behaviorQuality())
      )
    );
    ea.l.saveForRun(
      path = "../../Documenti/experiments/{name}/{startTime}/best-video-{run.solver.name}-{run.randomGenerator.seed:%04d}";
      of = ea.acc.lastBest();
      processor = f.composition(of = ea.f.solution(); then = s.taskVideoBuilder(task = s.task.locomotion(duration = 15)))
    )
  ]
)
